**Create `src/components/alias/AliasPenScene.tsx`:**

This is a simplified idle-only R3F scene. Per D-2: continuous idle float + gentle Y rotation, always-on, NOT scroll-coupled. Per D-3: useFrame + refs only, NO useState in animation hot paths.

```tsx
'use client';

import { useGLTF, ContactShadows } from '@react-three/drei';
import { useFrame } from '@react-three/fiber';
import { useRef, useMemo } from 'react';
import { Group, Object3D } from 'three';
import { aliasSceneConfig } from '@/lib/three/aliasSceneConfig';
import type { DeviceTier } from '@/hooks/useDeviceCapabilities';

type AliasPenSceneProps = {
  reducedMotion: boolean;
  motionScale: number;
  tier: DeviceTier;
};

export default function AliasPenScene({ reducedMotion, motionScale, tier }: AliasPenSceneProps) {
  const penGroupRef = useRef<Group>(null);
  const gltf = useGLTF('/models/pen3D.glb');

  const penModel = useMemo(() => {
    const clone = gltf.scene.clone(true);
    clone.traverse((child) => {
      const obj = child as Object3D & { castShadow?: boolean; receiveShadow?: boolean };
      if ('castShadow' in obj) obj.castShadow = true;
      if ('receiveShadow' in obj) obj.receiveShadow = true;
    });
    return clone;
  }, [gltf.scene]);

  const [px, py, pz] = aliasSceneConfig.pen.position;
  const [rx, , rz] = aliasSceneConfig.pen.rotation;
  const showContactShadows = tier !== 'low';

  // Per D-2: continuous idle animation — NOT scroll-coupled
  // Per D-3: refs only in useFrame, no useState
  useFrame((state, delta) => {
    if (!penGroupRef.current || reducedMotion) return;
    const t = state.clock.elapsedTime;

    // Float: sin-wave Y oscillation
    penGroupRef.current.position.y =
      py +
      Math.sin(t * aliasSceneConfig.pen.idle.yFrequency) *
        aliasSceneConfig.pen.idle.yAmplitude *
        motionScale;

    // Slow continuous Y rotation — "display case" spin
    penGroupRef.current.rotation.y +=
      delta * aliasSceneConfig.pen.idle.rotationYSpeed * motionScale;
  });

  return (
    <>
      <ambientLight intensity={0.15} color="#ffffff" />
      <directionalLight position={[3, 2, 4]} intensity={1.4} color="#ffffff" />
      <directionalLight position={[-2, 1, 2]} intensity={0.35} color="#d0d0d0" />
      <pointLight position={[0, 2, -3]} intensity={0.4} color="#b0bec5" />
      {/* Subtle blue accent to match design system #007bff */}
      <pointLight position={[1.5, 0, 2]} intensity={0.3} color="#007bff" />

      <group
        ref={penGroupRef}
        position={[px, py, pz]}
        rotation={[rx, aliasSceneConfig.pen.rotation[1], rz]}
      >
        <primitive object={penModel} />
      </group>

      {showContactShadows && (
        <ContactShadows position={[0, -1.05, 0]} opacity={0.2} scale={5} blur={2} far={2.5} />
      )}
    </>
  );
}

useGLTF.preload('/models/pen3D.glb');
```

Key architectural decisions:
- NO scrollProgressRef, NO penTargetRef, NO pointer parallax, NO camera lerp (per D-2)
- Single useFrame with refs only (per D-3)
- `reducedMotion` check at top of useFrame — pen stays static when enabled
- Blue accent pointLight at [1.5, 0, 2] ties the 3D scene to the #007bff design system
- `useGLTF.preload` is safe — drei deduplicates if HeroScene already preloaded it

**Create `src/components/alias/AliasPenCanvas.tsx`:**

Follow exact pattern from HeroCanvas.tsx but simplified (no scroll/pose props).

```tsx
'use client';

import { Canvas } from '@react-three/fiber';
import { Suspense } from 'react';
import AliasPenScene from '@/components/alias/AliasPenScene';
import { aliasSceneConfig } from '@/lib/three/aliasSceneConfig';
import type { DeviceTier } from '@/hooks/useDeviceCapabilities';

type AliasPenCanvasProps = {
  reducedMotion: boolean;
  motionScale: number;
  maxDpr: number;
  tier: DeviceTier;
};

export default function AliasPenCanvas({
  reducedMotion,
  motionScale,
  maxDpr,
  tier,
}: AliasPenCanvasProps) {
  const [cx, cy, cz] = aliasSceneConfig.camera.position;

  return (
    <div className="absolute inset-0" aria-hidden="true">
      <Canvas
        dpr={[1, maxDpr]}
        frameloop={reducedMotion ? 'demand' : 'always'}
        gl={{ antialias: true, alpha: true, powerPreference: 'high-performance' }}
        camera={{
          position: [cx, cy, cz],
          fov: aliasSceneConfig.camera.fov,
          near: 0.1,
          far: 100,
        }}
        onCreated={({ camera }) => camera.lookAt(0, 0, 0)}
      >
        <Suspense fallback={null}>
          <AliasPenScene
            reducedMotion={reducedMotion}
            motionScale={motionScale}
            tier={tier}
          />
        </Suspense>
      </Canvas>
    </div>
  );
}
```

Key decisions:
- `dpr={[1, maxDpr]}` where maxDpr ≤ 1.5 from useDeviceCapabilities (per D-3)
- `frameloop="demand"` when reducedMotion is true — freezes render loop (Phase 08 pattern)
- `aria-hidden="true"` on Canvas wrapper — 3D pen is decorative (Phase 07 decision)
- `alpha: true` — transparent background so page's #050a14 shows through
- No scroll/pointer props — this Canvas is simpler than HeroCanvas by design

Create the responsive two-column layout shell. Per D-1: glass form LEFT + 3D pen RIGHT. Per D-7: at 375px, hide 3D pen entirely, single-column.

```tsx
'use client';

import dynamic from 'next/dynamic';
import useDeviceCapabilities from '@/hooks/useDeviceCapabilities';
import useReducedMotion from '@/hooks/useReducedMotion';

const AliasPenCanvas = dynamic(
  () => import('@/components/alias/AliasPenCanvas'),
  { ssr: false, loading: () => null }
);

type AliasLayoutProps = {
  children: React.ReactNode;
};

export default function AliasLayout({ children }: AliasLayoutProps) {
  const reducedMotion = useReducedMotion();
  const { tier, supportsWebGL, recommendedMaxDpr } = useDeviceCapabilities();

  // motionScale per tier — same mapping as HeroSection
  const motionScale = tier === 'low' ? 0.5 : tier === 'medium' ? 0.8 : 1;

  return (
    <main className="mx-auto grid min-h-screen w-full max-w-7xl grid-cols-1 items-center gap-0 px-6 pt-28 pb-12 md:grid-cols-2 md:gap-8">
      {/* Left column: glass panel form content (always visible) */}
      <div className="w-full max-w-2xl">
        {children}
      </div>

      {/* Right column: 3D pen — hidden below md (per D-7), hidden if no WebGL */}
      <div
        className="relative hidden h-[500px] md:block"
        aria-hidden="true"
      >
        {supportsWebGL && (
          <AliasPenCanvas
            reducedMotion={reducedMotion}
            motionScale={motionScale}
            maxDpr={recommendedMaxDpr}
            tier={tier}
          />
        )}
      </div>
    </main>
  );
}
```

Key architectural decisions:
- `next/dynamic({ ssr: false })` on AliasPenCanvas — same pattern as HeroSection (Phase 02 decision)
- `hidden md:block` on Canvas column — CSS-first responsive hiding (per D-7). No JS hydration mismatch: the div is always in the DOM but CSS hides it below 768px. The `supportsWebGL` check adds a JS-based fallback for devices without WebGL.
- `h-[500px]` on Canvas container — explicit height prevents the "Canvas height = 0" pitfall from research RQ7. The `absolute inset-0` inside AliasPenCanvas needs a sized parent.
- `max-w-7xl` (wider than existing `max-w-6xl`) to give the two-column layout breathing room.
- `grid grid-cols-1 md:grid-cols-2` for clean responsive switching.
- Form column has `max-w-2xl` to prevent stretching on wide screens.
- `items-center` vertically centers both columns for visual balance.
- No `useMediaQuery('(min-width: 768px)')` JS check needed — CSS `hidden md:block` handles responsive hiding without hydration risk. The `supportsWebGL` boolean only gates the R3F Canvas instantiation (saves resources on non-WebGL devices).
